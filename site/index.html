Package.swift 
.package(
    url: "https://github.com/killa-kyle/iosUI-sdk.git",
    from: "0.2.1"
)
// A choice list with a custom row.
@State private var choice =
    Selection<String>(mode: .multiple)

BehaviorChoiceList(
    ["SwiftUI", "UIKit"],
    selection: $choice,
    label: { $0 }
) { name, selected in
    Label(name, systemImage:
        selected ? "checkmark" : "circle")
}
